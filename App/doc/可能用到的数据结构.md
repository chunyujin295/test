    UNKNOW = -1,
    TEXT_TO_MODEL,          // POST /tasks/text-to-model      文字→带纹理模型
    TEXT_TO_MESH,           // POST /tasks/text-to-mesh       文字→纯白模
    IMAGE_TO_MODEL,         // POST /tasks/image-to-model     图片→带纹理模型
    IMAGE_TO_MESH,          // POST /tasks/image-to-mesh      图片→纯白模
    TEXTURE_MODEL,          // POST /tasks/texture-model      图片+模型→纹理
    TEXT_TO_TEXTURE,        // POST /tasks/text-to-texture    文字+模型→纹理
    MODEL_PREVIEW_RENDER,   // POST /tasks/model-preview-render  模型→预览图

    CONVERT_MODEL_FORMAT,   // POST /tasks/convert-model-format  格式转换
    IMAGE_GENERATION,       // POST /tasks/image-generation      文字→图片

struct GenerationParams {
    // ── 任务类型 ──
    GenTaskCategory category = GenTaskCategory::TEXT_TO_3D;
    GenTaskSubType  sub_type = GenTaskSubType::TEXT_TO_MODEL;

    // ── 文字输入 ──
    std::optional<std::string> prompt;           // 正向提示词 [T2M, T2G, T2T, IMG]
    std::optional<std::string> negative_prompt;  // 反向提示词 [T2M, T2G, IMG]
    std::optional<std::string> style;            // 风格        [IMG]

    // ── 素材 ──
    std::optional<AssetRef>    image_file;       // 输入图片 [I2M, I2G, TxM]
    std::optional<AssetRef>    model_file;       // 输入模型 [TxM, T2T, MPR, RMS, CVT]

    // ── 生成控制 ──
    std::optional<std::string> model_version;    // 模型版本     [全部]
    std::optional<int>         polygon_limit;    // 面数上限     [T2M, T2G, I2M, I2G]
    std::optional<int>         texture_size;     // 纹理分辨率   [T2M, I2M, TxM]

    // ── 预览渲染 [MPR] ──
    std::optional<std::string>              render_mode;   // "single" / "multi_angle"
    std::optional<int>                      image_count;   // 出图张数
    std::optional<std::vector<std::string>> camera_angles; // 相机角度
    std::optional<std::string>              preset_name;   // 光照预设

    // ── 重网格 [RMS] ──
    std::optional<int>         target_poly_count;   // 目标面数

    // ── 格式转换 [CVT] ──
    std::optional<std::string> output_format;       // "glb"/"gltf"/"obj"/"fbx"/"stl"/"ply"
};
